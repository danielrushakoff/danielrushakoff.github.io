1. Project Overview & Objectives

I engineered a dual-sensor biomedical system utilizing an AD8232 for ECG data acquisition and a MAX30102 for pulse oximetry. The primary objective was to capture and monitor the live ECG P-QRS-T waveform accurately. To resolve critical 60 Hz AC mains interference disrupting the signal from wall power, I implemented custom C++ DSP algorithms to aggressively filter the noise while preserving high-frequency cardiac signal components.

2. System Architecture / Block Diagram

The system routes analog biomedical signals through a high-precision sampling pipeline, utilizing a hardware timer to maintain a strict 1 kHz sampling rate to guarantee DSP filter performance before telemetry processing.

3. Wireless IoT Firmware (Battery Powered)

After migrating the Arduino Uno R4 WiFi to a 9V battery power supply, the 60 Hz AC mains interference was eliminated. This finalized firmware integrates both the AD8232 and MAX30102 sensors wirelessly, transmitting live telemetry directly to the Arduino IoT Cloud Dashboard without the need for the DSP filter.

FspTimer ecgTimer;
const uint16_t ECG_BUFFER_SIZE = 512;
const uint16_t ECG_BUFFER_MASK = ECG_BUFFER_SIZE - 1;
volatile uint16_t ecgBuffer[ECG_BUFFER_SIZE];

volatile uint16_t ecgHead = 0;

void ecgTimerISR(timer_callback_args_t *p_args) {
  (void)p_args;
  uint32_t seq = ++isrTickCounter;
  uint16_t sample = (uint16_t)analogRead(PIN_ECG);
  uint16_t nextHead = (ecgHead + 1) & ECG_BUFFER_MASK;
  if (nextHead != ecgTail) {
    ecgBuffer[ecgHead] = sample;
    ecgSeqBuffer[ecgHead] = seq;
    ecgHead = nextHead;
  }
}

bool startEcgTimer(float freqHz) {
  uint8_t timerType = GPT_TIMER;
  int8_t timerChannel = FspTimer::get_available_timer(timerType);
  if (timerChannel < 0) {
    timerType = AGT_TIMER;
    timerChannel = FspTimer::get_available_timer(timerType);
  }
  if (timerChannel < 0) return false;

  if (!ecgTimer.begin(TIMER_MODE_PERIODIC, timerType, (uint8_t)timerChannel, freqHz, 0.0f, ecgTimerISR)) return false;
  if (!ecgTimer.setup_overflow_irq(12)) return false;
  if (!ecgTimer.open()) return false;
  if (!ecgTimer.start()) return false;
  return true;
}

// ==========================================
//                  LOOP
// ==========================================
void loop() {
  
  // ==========================================
  //      TASK 1: ECG LOGIC (Priority)
  // ==========================================
  uint8_t processed = 0;
  while (ecgTail != ecgHead && processed < 64) {
    uint16_t rawSample = ecgBuffer[ecgTail];
    uint32_t sampleSeq = ecgSeqBuffer[ecgTail];
    ecgTail = (ecgTail + 1) & ECG_BUFFER_MASK;
    processed++;

    if (prevSampleSeq != 0 && sampleSeq != (prevSampleSeq + 1)) {
      belowThreshold = true;
      beat_old = sampleSeq;
    }
    prevSampleSeq = sampleSeq;

    if ((digitalRead(PIN_LO_POS) == 1) || (digitalRead(PIN_LO_NEG) == 1)) {
      // Leads off
      peakValueInWindow = 512; 
      belowThreshold = true;
      beatIndex = 0; 
      ecg_BPM = 0;
      
      // Reset buffer
      for (int i = 0; i < BEATS_TO_AVERAGE; i++) beats[i] = 0;
      beat_old = sampleSeq; 
    }
    else {
      inByte = (float)rawSample;
      
      // Peak Hold
      if (inByte > peakValueInWindow) {
        peakValueInWindow = (int)inByte;
      }
      
      // BPM Math
      if (inByte > THRESHOLD && belowThreshold == true) {
        calculateECGBPM(sampleSeq); 
        belowThreshold = false;
      }
      else if (inByte < THRESHOLD) {
        belowThreshold = true; 
      }
    }
  }

  // ==========================================
  //      TASK 2: MAX30102 LOGIC
  // ==========================================
  particleSensor.check();
  while (particleSensor.available()) {
    redValue = particleSensor.getFIFORed();
    irValue  = particleSensor.getFIFOIR();
    particleSensor.nextSample();
    maxSampleCounter++;

    if (irValue > 50000) { 
      irBuffer[bufferIndex]  = (uint32_t)irValue;
      redBuffer[bufferIndex] = (uint32_t)redValue;
      bufferIndex++;

      if (bufferIndex >= BUFFER_LENGTH) {
        maxim_heart_rate_and_oxygen_saturation(
          irBuffer, BUFFER_LENGTH, redBuffer,
          &spo2, &spo2Valid, &spo2HR, &spo2HRValid
        );
        bufferIndex = 0;
      }

      if (checkForBeat(irValue) == true) {
        uint32_t deltaSamples = maxSampleCounter - lastBeatSample;
        lastBeatSample = maxSampleCounter;

        if (deltaSamples > 0) {
          max_BPM = (60.0f * MAX_FIFO_RATE_HZ) / (float)deltaSamples;

          if (max_BPM < 255 && max_BPM > 20) {
            rates[rateSpot++] = (byte)max_BPM;
            rateSpot %= RATE_SIZE;
            if (beatAvg < 20) {
               for (byte x = 0; x < RATE_SIZE; x++) rates[x] = (byte)max_BPM;
            }
            long sum = 0;
            for (byte x = 0; x < RATE_SIZE; x++) sum += rates[x];
            beatAvg = sum / RATE_SIZE;
          }
          cloudBeat = true;
        }
      }
      if (cloudBeat && (maxSampleCounter - lastBeatSample) > 3) {
        cloudBeat = false;
      }
    } 
    else {
      beatAvg = 0;
      max_BPM = 0;
      spo2 = 0;
      spo2Valid = 0;
      bufferIndex = 0;
      lastBeatSample = maxSampleCounter;
      for (byte x = 0; x < RATE_SIZE; x++) rates[x] = 0;
    }
  }

  // ==========================================
  //      TASK 3: CLOUD & SERIAL UPDATES
  // ==========================================
  if (millis() - lastPrintTime >= PRINT_SPEED) {
    lastPrintTime = millis();
    
    // Update Variables
    ecg_signal = peakValueInWindow;
    bpm_value = ecg_BPM;
    
    cloudBPM = (int)max_BPM;
    cloudAvgBPM = beatAvg;
    cloudIR = irValue;
    cloudSpO2 = (spo2Valid == 1) ? spo2 : 0;

    // --- SERIAL PLOTTER ---
    Serial.print("ECG:");
    Serial.print(peakValueInWindow);      
    Serial.print(",");
    Serial.print("ECG_BPM:");
    Serial.print(ecg_BPM);   
    Serial.print(",");
    Serial.print("MAX_IR:");
    Serial.print(irValue);     
    Serial.print(",");
    Serial.print("MAX_BPM:");
    Serial.println(beatAvg);  

    peakValueInWindow = (int)inByte; 
    
    ArduinoCloud.update(); 
  }
}

// ==========================================
//          HELPER FUNCTIONS
// ==========================================

void calculateECGBPM (uint32_t beat_new) {  
  if (beat_old == 0) {
    beat_old = beat_new;
    return;
  }

  uint32_t diff = beat_new - beat_old;
  
  if (diff > MIN_TIME_BETWEEN_BEATS) {
    float currentBPM = 60000.0f / (float)diff; 
    
    if (currentBPM > 30 && currentBPM < 220) {
      beats[beatIndex] = currentBPM; 
      
      if (ecg_BPM < 20) {
         for (int i = 0; i < BEATS_TO_AVERAGE; i++) beats[i] = currentBPM;
      }
      
      float total = 0.0;
      for (int i = 0; i < BEATS_TO_AVERAGE; i++){
        total += beats[i];
      }
      
      ecg_BPM = int(total / BEATS_TO_AVERAGE);
      beat_old = beat_new;
      beatIndex = (beatIndex + 1) % BEATS_TO_AVERAGE; 
    }
    else if (diff > 2500) { 
       beat_old = beat_new;
    }
  }
}

4. DSP Filter Code (Wired Configuration)

During the initial wired testing phase, the system was powered via a 5V USB connection to a laptop plugged into a wall outlet, which introduced significant 60 Hz AC noise. To target the exact 60 Hz interference without applying a broad low-pass filter that would attenuate the QRS complex, I designed a second-order IIR Biquad Notch Filter. The filter was modeled in MATLAB with zeros placed on the unit circle at 60 Hz in the z-plane, then converted into the C++ difference equation below for real-time sample processing.

#include <Wire.h>
#include <cmath>
#include "MAX30105.h"
#include "heartRate.h"
#include "spo2_algorithm.h"
#include "ArduinoIoTCloud.h"
#include "Arduino_ConnectionHandler.h"
#include <FspTimer.h>

// ==========================================
//       60 Hz IIR BIQUAD NOTCH FILTER
// ==========================================
class NotchFilter {
public:
  using sample_t = float;

  NotchFilter(sample_t sampleRateHz, sample_t notchFreqHz = 60.0f, sample_t Q = 30.0f) {
    configure(sampleRateHz, notchFreqHz, Q);
  }

  void configure(sample_t sampleRateHz, sample_t notchFreqHz, sample_t Q) {
    const sample_t w0 = 2.0f * static_cast<sample_t>(M_PI) * notchFreqHz / sampleRateHz;
    const sample_t alpha = std::sin(w0) / (2.0f * Q);
    const sample_t cosw0 = std::cos(w0);

    const sample_t a0 = 1.0f + alpha;
    b0_ = 1.0f / a0;
    b1_ = -2.0f * cosw0 / a0;
    b2_ = 1.0f / a0;
    a1_ = -2.0f * cosw0 / a0;
    a2_ = (1.0f - alpha) / a0;

    reset();
  }

  void reset(sample_t dcBaseline = 310.0f) {
    x1_ = x2_ = y1_ = y2_ = dcBaseline;
  }

  sample_t process(sample_t x) {
    const sample_t y = b0_ * x + b1_ * x1_ + b2_ * x2_ - a1_ * y1_ - a2_ * y2_;
    x2_ = x1_;
    x1_ = x;
    y2_ = y1_;
    y1_ = y;
    return y;
  }

private:
  sample_t b0_ = 1, b1_ = 0, b2_ = 0, a1_ = 0, a2_ = 0;
  sample_t x1_ = 310.0f, x2_ = 310.0f, y1_ = 310.0f, y2_ = 310.0f;
};

  uint32_t diff = beat_new - beat_old;
  
  if (diff > MIN_TIME_BETWEEN_BEATS) {
    float currentBPM = 60000.0f / (float)diff; 
    
    if (currentBPM > 30 && currentBPM < 220) {
      beats[beatIndex] = currentBPM; 
      
      if (ecg_BPM < 20) {
         for (int i = 0; i < BEATS_TO_AVERAGE; i++) beats[i] = currentBPM;
      }
      
      float total = 0.0;
      for (int i = 0; i < BEATS_TO_AVERAGE; i++){
        total += beats[i];
      }
      
      ecg_BPM = int(total / BEATS_TO_AVERAGE);
      beat_old = beat_new;
      beatIndex = (beatIndex + 1) % BEATS_TO_AVERAGE; 
    }
    else if (diff > 2500) { 
       beat_old = beat_new;
    }
  }
}
