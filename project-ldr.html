1. Overview & Objectives

This project involved designing a wireless control system that measures ambient light intensity via a Light-Dependent Resistor (LDR) on a transmitter node, and relays that data via RF to a receiver node. The receiver scales the data to modulate the brightness of an LED while communicating real-time telemetry to a custom LabVIEW dashboard.

Transmitter Code (Lab Report Draft)

Draft sketch for the transmitter node responsible for reading the analog LDR data and broadcasting it via the nRF24L01+ transceiver.

// ---------------------------------------------------------------- //
//           ECE3709L LDR TRANSMITTER - LAB REPORT CODE             //
// ---------------------------------------------------------------- //
#include <SPI.h>
#include <RF24.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

// ---- Hardware mapping ----
const int PIN_LDR = A0;
const uint8_t RF_CE_TX = 7; // CE
const uint8_t RF_CSN_TX = 8; // CSN

// I2C LCD: try 0x27 first; change to 0x3F if needed
LiquidCrystal_I2C lcd(0x27, 16, 2);
RF24 radio(RF_CE_TX, RF_CSN_TX);

// 5-byte pipe address (Must match RX)
const byte pipeAddr[6] = "ABCD"; // Changed from "L7TX1" to match your Receiver

void setup() {
  // LCD
  lcd.init();
  lcd.backlight();
  lcd.setCursor(0,0); lcd.print("LDR Wireless TX");
  lcd.setCursor(0,1); lcd.print("Init...");

  // RF
  radio.begin();
  // Slower rate improves range/robustness in labs
  radio.setDataRate(RF24_250KBPS);
  radio.setRetries(5, 15); // delay,count
  radio.openWritingPipe(pipeAddr);
  radio.stopListening(); // TX role
  delay(400);
  lcd.setCursor(0,1); lcd.print("Ready"); 
}

void loop() {
  int raw = analogRead(PIN_LDR); // 0..1023
  uint8_t level = raw / 100; // ~0..10 compact

  // Display
  lcd.setCursor(0,0); lcd.print("LDR: "); 
  lcd.setCursor(5,0); lcd.print((int)level);
  lcd.setCursor(0,1); lcd.print("Send: "); 
  lcd.setCursor(6,1); lcd.print((int)level);

  // Transmit one byte
  radio.write(&level, sizeof(level));
  delay(100);
}
// Note: Remember to replace any < or > brackets with < and >
