1. Overview & Objectives

This project demonstrates a closed-loop wireless motor control system utilizing a custom-tuned PID algorithm. The objective was to design a robust transmitter and receiver architecture capable of eliminating steady-state error and preventing integral windup during continuous motor operation, while relaying real-time telemetry data to a custom LabVIEW GUI for monitoring and analysis.

Transmitter Code

// ---------------------------------------------------------------- //
    //           ECE3709L PID TRANSMITTER - LABVIEW & TX SIDE           //
    // ---------------------------------------------------------------- //
    #include <SPI.h>
    #include <RF24.h>
    #include <Wire.h>
    #include <LiquidCrystal_I2C.h>

    // -------------------- Pin Definitions --------------------
    #define CE_PIN 9
    #define CSN_PIN 10

    // -------------------- LCD Setup --------------------
    LiquidCrystal_I2C lcd(0x27, 16, 2); // 16x2 I2C LCD

    // -------------------- RF24 Setup --------------------
    RF24 radio(CE_PIN, CSN_PIN);
    const byte address[6] = "3Node"; // 5-byte address

    // -------------------- Global Variables --------------------
    char cmdBuf[16]; // Matched to receiver's 16-byte payload buffer

    // -------------------- Setup --------------------
    void setup() {
      // Initialize Serial for LabVIEW communication
      Serial.begin(9600);
      while (!Serial);
      Serial.println("TX: Serial initialized");

      // Initialize LCD
      lcd.init();
      lcd.backlight();
      lcd.setCursor(0, 0);
      lcd.print("TX Ready");
      lcd.setCursor(0, 1);
      lcd.print("Waiting cmd...");

      // Initialize NRF24
      if (!radio.begin()) {
        Serial.println("TX: RF module failed");
        lcd.setCursor(0, 1);
        lcd.print("RF FAIL        ");
        while (1); // halt
      }
      radio.setPALevel(RF24_PA_LOW);       // Lab distance, avoid noise
      radio.setDataRate(RF24_1MBPS);       // Reliable default
      radio.openWritingPipe(address);
      radio.stopListening();               // Transmitter mode
      Serial.println("TX: RF Ready");
    }

    // -------------------- Main Loop --------------------
    void loop() {
      // Check if data is available from LabVIEW GUI
      if (Serial.available()) {
        int len = Serial.readBytesUntil('\n', cmdBuf, sizeof(cmdBuf) - 1);
        cmdBuf[len] = '\0'; // Null-terminate the string

        // Convert to String for easier parsing and uppercase formatting
        String cmdStr = String(cmdBuf);
        cmdStr.trim();
        cmdStr.toUpperCase();
        cmdStr.toCharArray(cmdBuf, sizeof(cmdBuf));

        // Determine if command is valid for the PID Receiver
        bool validCmd = false;
        if (cmdStr == "FWD" || cmdStr == "REV" || cmdStr == "STOP" ||
            cmdStr == "PID:ON" || cmdStr == "PID:OFF" || cmdStr.startsWith("SET:")) {
            validCmd = true;
        }

        if (!validCmd) {
          Serial.print("TX: Bad cmd '");
          Serial.print(cmdBuf);
          Serial.println("'");
          lcd.setCursor(0, 1);
          lcd.print("BAD CMD        ");
          return;
        }

        // Update LCD Output
        lcd.clear();
        lcd.setCursor(0, 0);
        lcd.print("TX: ");

        if (cmdStr.startsWith("SET:")) {
          lcd.print("SET RPM");
          lcd.setCursor(0, 1);
          lcd.print(cmdStr.substring(4)); // Display just the target RPM number
        } else {
          lcd.print(cmdStr); // Display FWD, STOP, PID:ON, etc.
        }

        // Transmit via RF to Motor Node
        bool ok = radio.write(cmdBuf, sizeof(cmdBuf));
        if (ok) {
          Serial.print("TX: Sent '");
          Serial.print(cmdBuf);
          Serial.println("'");
        } else {
          Serial.print("TX: Failed '");
          Serial.print(cmdBuf);
          Serial.println("'");
          lcd.setCursor(0, 1);
          lcd.print("TX FAIL        ");
        }

        delay(50); // Small debounce for RF stability
      }
    }
    // Note: Remember to replace any < or > brackets with < and >
    // For example: #include <Wire.h>

Receiver Code

// ---------------------------------------------------------------- //
    //           ECE3709L PID RECEIVER - MOTOR & SENSOR SIDE            //
    // ---------------------------------------------------------------- //
    #include <SPI.h>
    #include <RF24.h>
    #include <Wire.h>
    #include <LiquidCrystal_I2C.h>

    // -------------------- PIN DEFINITIONS --------------------
    // NOTE: Motor pins moved to 4 & 5 to free up Pin 2 for the Sensor Interrupt!
    const uint8_t CE_PIN  = 7;
    const uint8_t CSN_PIN = 8;
    const uint8_t EN1_PIN = 9;  // PWM Enable
    const uint8_t IN1_PIN = 4;  // CHANGED from 3
    const uint8_t IN2_PIN = 5;  // CHANGED from 2

    // PID Sensor (LM393 Optical Encoder)
    const uint8_t SPEED_SENSOR_PIN = 2; // Must be Pin 2 or 3 for interrupts

    // -------------------- GLOBAL VARIABLES --------------------
    RF24 radio(CE_PIN, CSN_PIN);
    LiquidCrystal_I2C lcd(0x27, 20, 4);
    const byte address[6] = "3Node";
    char payload[16]; // Increased buffer slightly for longer commands

    // --- PID CONSTANTS (Tune these!) ---
    // See Lab Manual Section 9.A.1 [cite: 321]
    double Kp = 2.0;   // Proportional gain
    double Ki = 0.5;   // Integral gain
    double Kd = 0.1;   // Derivative gain

    // --- PID VARIABLES ---
    bool pidEnabled = false;       // Manual Mode by default [cite: 121]
    double setpointRPM = 0;        // Desired Speed
    double currentRPM = 0;         // Measured Speed (Process Variable)
    double pwmOutput = 0;          // Control Output (0-255)
    double previousError = 0;
    double integral = 0;
    unsigned long lastPIDTime = 0;
    const int PID_INTERVAL = 100;  // 100ms Sampling Loop [cite: 181]

    // --- SPEED MEASUREMENT VARIABLES ---
    volatile unsigned long pulseCount = 0; // "volatile" because it changes in interrupt
    const int ENCODER_SLOTS = 20;          // Standard encoder disk slots [cite: 288]

    // -------------------- INTERRUPT ROUTINE --------------------
    // This function runs automatically every time the sensor sees a slot
    void countPulse() {
      pulseCount++;
    }

    // -------------------- MOTOR FUNCTIONS --------------------
    void motorDrive(int pwm, bool reverse) {
      // Constrain PWM to safe limits
      pwm = constrain(pwm, 0, 255);

      if (pwm == 0) {
        digitalWrite(IN1_PIN, LOW);
        digitalWrite(IN2_PIN, LOW);
        analogWrite(EN1_PIN, 0);
      } else if (!reverse) {
        digitalWrite(IN1_PIN, HIGH);
        digitalWrite(IN2_PIN, LOW);
        analogWrite(EN1_PIN, pwm);
      } else {
        digitalWrite(IN1_PIN, LOW);
        digitalWrite(IN2_PIN, HIGH);
        analogWrite(EN1_PIN, pwm);
      }
    }

    // -------------------- SETUP --------------------
    void setup() {
      Serial.begin(9600);

      // 1. Init Pins
      pinMode(IN1_PIN, OUTPUT);
      pinMode(IN2_PIN, OUTPUT);
      pinMode(EN1_PIN, OUTPUT);
      pinMode(SPEED_SENSOR_PIN, INPUT_PULLUP); // Sensor Input

      // 2. Attach Interrupt for RPM Counting [cite: 292]
      attachInterrupt(digitalPinToInterrupt(SPEED_SENSOR_PIN), countPulse, RISING);

      // 3. Init LCD
      Wire.begin();
      lcd.init();
      lcd.backlight();
      lcd.setCursor(0,0);
      lcd.print("PID System Init");

      // 4. Init Radio
      if (!radio.begin()) {
        lcd.setCursor(0,1);
        lcd.print("Radio Fail");
        while(1);
      }
      radio.setPALevel(RF24_PA_LOW);
      radio.setDataRate(RF24_1MBPS);
      radio.openReadingPipe(0, address);
      radio.startListening();

      // Flush buffer
      delay(100);
      while(radio.available()) {
        char junk[8];
        radio.read(&junk, sizeof(junk));
      }
    }

    // -------------------- MAIN LOOP --------------------
    void loop() {
      unsigned long currentTime = millis();

      // --- A. READ WIRELESS COMMANDS ---
      if (radio.available()) {
        memset(payload, 0, sizeof(payload));
        radio.read(payload, sizeof(payload));

        String cmd = String(payload);
        cmd.trim();
        cmd.toUpperCase();

        if (cmd.length() > 0) {
          Serial.print("RX: "); Serial.println(cmd);
          parseCommand(cmd); // Handle FWD, PID, SET, etc.
        }
      }

      // --- B. RUN PID LOOP (Time-Based) ---
      if (currentTime - lastPIDTime >= PID_INTERVAL) {
        // 1. Calculate RPM from pulses
        // RPM = (Pulses / Slots) * (60000ms / Interval_ms)
        // We disable interrupts briefly to read the volatile variable safely
        noInterrupts();
        unsigned long pulses = pulseCount;
        pulseCount = 0; // Reset counter for next interval
        interrupts();

        currentRPM = (float(pulses) / ENCODER_SLOTS) * (60000.0 / PID_INTERVAL);

        // 2. Compute Output
        if (pidEnabled) {
          pwmOutput = computePID(setpointRPM, currentRPM);
        }
        // If Manual Mode, pwmOutput is set directly by commands (parsing section)

        // 3. Drive Motor
        // Note: Direction is handled by global state or assumed FWD for PID in this lab
        motorDrive((int)pwmOutput, false);

        // 4. Update Display
        updateLCD();

        lastPIDTime = currentTime;
      }
    }

    // -------------------- PID CALCULATION --------------------
    double computePID(double setpoint, double measured) {
      // [cite: 201] Calculate Error
      double error = setpoint - measured;

      // [cite: 202] Proportional Term
      double P = Kp * error;

      // [cite: 203] Integral Term (with Anti-Windup)
      integral += error * (PID_INTERVAL / 1000.0);

      // Anti-Windup: Stop growing integral if we are already maxed out [cite: 371]
      if (integral > 255) integral = 255;
      if (integral < -255) integral = -255;

      double I = Ki * integral;

      // [cite: 205] Derivative Term
      // D = Kd * (error - previousError) / dt
      double derivative = (error - previousError) / (PID_INTERVAL / 1000.0);
      double D = Kd * derivative;

      previousError = error;

      // [cite: 207] Total Output
      double output = P + I + D;

      // [cite: 208] Constrain output
      return constrain(output, 0, 255);
    }

    // -------------------- COMMAND PARSING --------------------
    void parseCommand(String cmd) {
      // Manual Commands
      if (cmd == "STOP") {
        pidEnabled = false;
        pwmOutput = 0;
        setpointRPM = 0;
        integral = 0; // Reset accumulated error
      }
      else if (cmd == "FWD") {
        pidEnabled = false;
        pwmOutput = 200; // Default manual speed
      }
      else if (cmd == "REV") {
        pidEnabled = false;
        // For PID lab, we usually only control speed in FWD, but this allows basic check
        motorDrive(200, true);
      }

      // PID Commands (Format: "SET:1500" or "PID:ON")
      else if (cmd.startsWith("SET:")) {
        // Extract number after ":"
        setpointRPM = cmd.substring(4).toInt();
        Serial.print("Setpoint Updated: "); Serial.println(setpointRPM);
      }
      else if (cmd == "PID:ON") {
        pidEnabled = true;
        integral = 0; // Reset integral on fresh start
      }
      else if (cmd == "PID:OFF") {
        pidEnabled = false;
        pwmOutput = 0;
      }
    }

    // -------------------- LCD DISPLAY --------------------
    void updateLCD() {
      lcd.setCursor(0, 0);
      if (pidEnabled) lcd.print("Mode: PID   ");
      else            lcd.print("Mode: MANUAL");

      lcd.setCursor(0, 1);
      lcd.print("Set:");
      lcd.print((int)setpointRPM);
      lcd.print(" RPM:");
      lcd.print((int)currentRPM); // Display Feedback [cite: 215]

      lcd.setCursor(0, 2); // 20x4 LCD has rows 2 & 3
      lcd.print("PWM: ");
      lcd.print((int)pwmOutput);
    }
    // Note: Remember to replace any < or > brackets with < and >
